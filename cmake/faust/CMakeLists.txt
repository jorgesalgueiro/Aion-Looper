cmake_minimum_required(VERSION 3.22)
project(AionFaustDependency LANGUAGES C CXX)
if(NOT EXISTS "${AION_FAUST_SOURCE_DIR}/build/CMakeLists.txt")
    message(FATAL_ERROR "AION_FAUST_SOURCE_DIR must contain verified Faust 2.88.0 sources")
endif()
file(READ "${AION_FAUST_SOURCE_DIR}/build/CMakeLists.txt" version_source)
if(NOT version_source MATCHES "set \\(VERSION 2\\.88\\.0\\)")
    message(FATAL_ERROR "Only pinned Faust 2.88.0 is supported")
endif()
# Isolated compiler build: no LLVM/JIT, external compiler backend, OSC/HTTP or
# system install. The optional host target owns the only public entry points.
foreach(backend AS C CPP CMAJOR CSHARP DLANG FIR INTERP_COMP JAVA JULIA JSFX LINEN LLVM NNX OLDCPP RUST TEMPLATE WASM CODEBOX JAX VHDL)
    set(${backend}_BACKEND OFF CACHE STRING "Disabled Faust backend" FORCE)
endforeach()
set(INTERP_BACKEND STATIC CACHE STRING "Portable Faust VM only" FORCE)
foreach(feature EXECUTABLE DYNAMIC OSC HTTP ITP EMCC WASM_GLUE WASMTIME LLVM)
    set(INCLUDE_${feature} OFF CACHE BOOL "Disabled Faust feature" FORCE)
endforeach()
set(INCLUDE_STATIC ON CACHE BOOL "Build interpreter static library" FORCE)
set(SELF_CONTAINED_LIBRARY ON CACHE BOOL "No system architecture search" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_LIST_DIR}/PatchInterpreter.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/PatchVmInstructions.cmake")
add_subdirectory("${AION_FAUST_SOURCE_DIR}/build" faust-build)
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(staticlib PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/lib/${config}")
endforeach()
set_target_properties(staticlib PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/Release")
file(WRITE "${CMAKE_BINARY_DIR}/aion-faust-runtime.txt" "2.88.0;silent-compute;trace-disabled;defined-vm-instructions\n")
